8. Coordenadas: decisión crítica del proyecto
En OpenSees usaremos normalmente:

X, Y → planta
Z    → vertical
Unity usa:

X → derecha
Y → arriba
Z → adelante
Para este proyecto adoptaremos inicialmente:

$$
X_U = X_{OS}
$$
 

$$
Y_U = Z_{OS}
$$
 

$$
Z_U = Y_{OS}
$$
 

Es decir:

OpenSees (X, Y, Z)
        ↓
Unity   (X, Z, Y)
Función única de conversión
Crear:

using UnityEngine;

public static class CoordinateMap
{
    public static Vector3 OsToUnity(float x, float y, float z)
    {
        return new Vector3(x, z, y);
    }

    public static Vector3 OsToUnity(Vector3 p)
    {
        return new Vector3(p.x, p.z, p.y);
    }

    public static Vector3 UnityToOs(Vector3 p)
    {
        return new Vector3(p.x, p.z, p.y);
    }
}
No repetir conversiones manuales dispersas como:

new Vector3(node.x, node.z, node.y)
en veinte scripts distintos.

Todo debe pasar por CoordinateMap.

12. ElementView.cs
Crear:

Assets/Scripts/View/ElementView.cs
using UnityEngine;

public class ElementView : MonoBehaviour
{
    public int ElementTag { get; private set; }
    public int NodeI { get; private set; }
    public int NodeJ { get; private set; }
    public string Kind { get; private set; }

    private float radius;

    public void Initialize(
        int elementTag,
        int nodeI,
        int nodeJ,
        string kind,
        float radius
    )
    {
        ElementTag = elementTag;
        NodeI = nodeI;
        NodeJ = nodeJ;
        Kind = kind;
        this.radius = radius;
    }

    public void SetEndpoints(Vector3 a, Vector3 b)
    {
        Vector3 d = b - a;
        float length = d.magnitude;

        if (length <= 1.0e-8f)
        {
            Debug.LogError(
                $"Element {ElementTag} has zero or near-zero length."
            );
            return;
        }

        transform.position = 0.5f * (a + b);

        transform.rotation =
            Quaternion.FromToRotation(
                Vector3.up,
                d.normalized
            );

        // Unity Cylinder primitive has height = 2
        transform.localScale =
            new Vector3(
                radius,
                0.5f * length,
                radius
            );
    }
}

18. Cargar un TextAsset
Guardar temporalmente:

Assets/DataSamples/model.json
Unity lo importa como un TextAsset.

Crear:

Assets/Scripts/Model/ModelLoader.cs
using System.Collections.Generic;
using UnityEngine;

public class ModelLoader : MonoBehaviour
{
    [SerializeField]
    private TextAsset modelJson;

    [SerializeField]
    private Transform modelRoot;

    [SerializeField]
    private float elementRadius = 0.05f;

    private readonly Dictionary<int, NodeData>
        nodes = new();

    private readonly Dictionary<int, ElementView>
        elements = new();

    private void Start()
    {
        LoadModel();
    }

    public void LoadModel()
    {
        if (modelJson == null)
        {
            Debug.LogError("modelJson is not assigned.");
            return;
        }

        ModelData data =
            JsonUtility.FromJson<ModelData>(
                modelJson.text
            );

        BuildNodeIndex(data);
        BuildElements(data);
    }

    private void BuildNodeIndex(ModelData data)
    {
        nodes.Clear();

        foreach (NodeData node in data.nodes)
        {
            if (!nodes.TryAdd(node.tag, node))
            {
                Debug.LogError(
                    $"Duplicated nodeTag: {node.tag}"
                );
            }
        }
    }

    private void BuildElements(ModelData data)
    {
        elements.Clear();

        foreach (ElementData element in data.elements)
        {
            if (!nodes.TryGetValue(
                element.i,
                out NodeData nodeI
            ))
            {
                Debug.LogError(
                    $"Element {element.tag}: " +
                    $"node I={element.i} does not exist."
                );

                continue;
            }

            if (!nodes.TryGetValue(
                element.j,
                out NodeData nodeJ
            ))
            {
                Debug.LogError(
                    $"Element {element.tag}: " +
                    $"node J={element.j} does not exist."
                );

                continue;
            }

            Vector3 a =
                CoordinateMap.OsToUnity(
                    nodeI.x,
                    nodeI.y,
                    nodeI.z
                );

            Vector3 b =
                CoordinateMap.OsToUnity(
                    nodeJ.x,
                    nodeJ.y,
                    nodeJ.z
                );

            GameObject go =
                GameObject.CreatePrimitive(
                    PrimitiveType.Cylinder
                );

            go.name =
                $"Element_{element.tag}";

            if (modelRoot != null)
            {
                go.transform.SetParent(modelRoot);
            }

            ElementView view =
                go.AddComponent<ElementView>();

            view.Initialize(
                element.tag,
                element.i,
                element.j,
                element.kind,
                elementRadius
            );

            view.SetEndpoints(a, b);

            if (!elements.TryAdd(
                element.tag,
                view
            ))
            {
                Debug.LogError(
                    $"Duplicated elementTag: " +
                    $"{element.tag}"
                );
            }
        }
    }
}

25. StructuralSelector.cs
using UnityEngine;
using UnityEngine.InputSystem;

public class StructuralSelector : MonoBehaviour
{
    [SerializeField]
    private Camera sceneCamera;

    [SerializeField]
    private LayerMask selectableLayers = ~0;

    public ElementView Selected { get; private set; }

    private void Update()
    {
        Mouse mouse = Mouse.current;

        if (mouse == null)
            return;

        if (!mouse.leftButton.wasPressedThisFrame)
            return;

        Vector2 mousePosition =
            mouse.position.ReadValue();

        Ray ray =
            sceneCamera.ScreenPointToRay(
                mousePosition
            );

        if (
            Physics.Raycast(
                ray,
                out RaycastHit hit,
                10000.0f,
                selectableLayers
            )
        )
        {
            ElementView view =
                hit.collider
                    .GetComponentInParent<ElementView>();

            if (view != null)
            {
                Select(view);
            }
        }
    }

    private void Select(ElementView view)
    {
        Selected = view;

        Debug.Log(
            $"Selected element: " +
            $"{view.ElementTag}, " +
            $"nodes {view.NodeI}-{view.NodeJ}"
        );
    }
}

28. Cámara orbital simple
Crear:

Assets/Scripts/Interaction/OrbitCamera.cs
using UnityEngine;
using UnityEngine.InputSystem;

public class OrbitCamera : MonoBehaviour
{
    [SerializeField]
    private Transform target;

    [SerializeField]
    private float distance = 20.0f;

    [SerializeField]
    private float rotateSpeed = 0.15f;

    [SerializeField]
    private float zoomSpeed = 0.01f;

    private float yaw = 45.0f;
    private float pitch = 30.0f;

    private void LateUpdate()
    {
        Mouse mouse = Mouse.current;

        if (mouse != null)
        {
            if (mouse.rightButton.isPressed)
            {
                Vector2 delta =
                    mouse.delta.ReadValue();

                yaw +=
                    delta.x * rotateSpeed;

                pitch -=
                    delta.y * rotateSpeed;

                pitch =
                    Mathf.Clamp(
                        pitch,
                        -85.0f,
                        85.0f
                    );
            }

            float scroll =
                mouse.scroll.ReadValue().y;

            distance *=
                Mathf.Exp(
                    -scroll * zoomSpeed
                );

            distance =
                Mathf.Clamp(
                    distance,
                    0.5f,
                    500.0f
                );
        }

        Quaternion rotation =
            Quaternion.Euler(
                pitch,
                yaw,
                0.0f
            );

        Vector3 offset =
            rotation *
            new Vector3(
                0.0f,
                0.0f,
                -distance
            );

        transform.position =
            target.position + offset;

        transform.LookAt(target);
    }
}

32. Helper mínimo para una línea

Assets/Scripts/View/LineFactory.cs
using UnityEngine;

public static class LineFactory
{
    public static LineRenderer CreateLine(
        Transform parent,
        Vector3[] points,
        float width,
        Material material
    )
    {
        GameObject go =
            new GameObject("Line");

        go.transform.SetParent(parent);

        LineRenderer lr =
            go.AddComponent<LineRenderer>();

        lr.positionCount =
            points.Length;

        lr.widthMultiplier =
            width;

        lr.useWorldSpace =
            true;

        lr.material =
            material;

        lr.SetPositions(points);

        return lr;
    }
}
Para varios puntos es preferible usar:

SetPositions(...)
en vez de llamar SetPosition(...) repetidamente.

35. Calcular área tributaria
Para un polígono en planta puede usarse la fórmula del cordón:

$$
A = \frac{1}{2} \left| \sum_i x_i y_{i+1} - x_{i+1} y_i \right|
$$
 

En C#:

using UnityEngine;

public static class PolygonMath
{
    public static float Area(
        Vector2[] vertices
    )
    {
        if (
            vertices == null ||
            vertices.Length < 3
        )
        {
            return 0.0f;
        }

        double sum = 0.0;

        for (
            int i = 0;
            i < vertices.Length;
            ++i
        )
        {
            int j =
                (i + 1) %
                vertices.Length;

            sum +=
                vertices[i].x *
                vertices[j].y;

            sum -=
                vertices[j].x *
                vertices[i].y;
        }

        return
            0.5f *
            Mathf.Abs((float)sum);
    }
}
36. Verificación de carga tributaria
Si:

q = kN/m²
A = m²
L = m
entonces:

$$
W = qA
$$
 

y, para una distribución uniforme idealizada:

$$
w = \frac{qA}{L}
$$
 

Unity puede calcularlo para mostrarlo, pero el mismo cálculo debe verificarse en Python/OpenSees.

41. Deformada
La posición gráfica deformada es:

$$
\mathbf{x}_{vis} = \mathbf{x}_0 + s\mathbf{u}
$$
 

donde:

\mathbf{x}_0: posición original;

\mathbf{u}: desplazamiento;

s: factor gráfico de amplificación.

Importante
El factor s no modifica el análisis.

Solo modifica la visualización.

42. Aplicar deformada a un elemento
Si conocemos:

Vector3 pI0;
Vector3 pJ0;

Vector3 uI;
Vector3 uJ;

float scale;
entonces:

Vector3 pI =
    pI0 +
    scale * uI;

Vector3 pJ =
    pJ0 +
    scale * uJ;

elementView.SetEndpoints(
    pI,
    pJ
);
43. Mantener geometría original separada de la deformada
No sobrescribir datos originales con desplazamientos.

Guardar:

positionOriginal
y calcular la deformada cada vez.

Esto permite:

deformed scale = 0
para volver inmediatamente al modelo original.

45. Superposición en Unity
Para cada nodo:

$$
\mathbf{u} = \lambda_G \mathbf{u}_G + \lambda_Q \mathbf{u}_Q + \lambda_X \mathbf{u}_{EX} + \lambda_Y \mathbf{u}_{EY}
$$
 

Ejemplo C#:

Vector3 Combine(
    Vector3 uG,
    Vector3 uQ,
    Vector3 uEX,
    Vector3 uEY,
    float lambdaG,
    float lambdaQ,
    float lambdaEX,
    float lambdaEY
)
{
    return
        lambdaG * uG +
        lambdaQ * uQ +
        lambdaEX * uEX +
        lambdaEY * uEY;
}

49. Dibujar un diagrama

donde:

c: escala gráfica;

\mathbf{d}: dirección visual asociada al diagrama.

Los puntos resultantes se dibujan con LineRenderer.

60. Punto dentro de polígono
using UnityEngine;

public static class PointInPolygon
{
    public static bool Contains(
        Vector2 p,
        Vector2[] polygon
    )
    {
        bool inside = false;

        int j =
            polygon.Length - 1;

        for (
            int i = 0;
            i < polygon.Length;
            ++i
        )
        {
            Vector2 pi = polygon[i];
            Vector2 pj = polygon[j];

            bool intersects =
                (
                    (pi.y > p.y) !=
                    (pj.y > p.y)
                )
                &&
                (
                    p.x <
                    (pj.x - pi.x) *
                    (p.y - pi.y) /
                    (pj.y - pi.y) +
                    pi.x
                );

            if (intersects)
                inside = !inside;

            j = i;
        }

        return inside;
    }
}
61. Una regla continua para repartir la carga
Supongamos que el panel tiene cuatro vigas receptoras.

Podemos definir una regla didáctica:

$$
w_i = \frac{1}{\max(d_i,\epsilon)}
$$
 

$$
\alpha_i = \frac{w_i}{\sum_j w_j}
$$
 

$$
P_i = \alpha_i P_{user}
$$
 

donde:

d_i: distancia del usuario a la viga i;

P_i: carga asignada.

Así, al acercarse a una viga:

su fracción aumenta
y siempre:

$$
\sum_i P_i = P_{user}
$$
